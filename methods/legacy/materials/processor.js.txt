  function blockLabel(method, headers, indexes, block) {
    if (method.id !== "leary_dmo_128") return "Профиль";
    const h=clean(headers[indexes[block*method.items]]);
    if (h.includes("идеаль")) return "Я-идеальное";
    if (h.includes("реаль")) return "Я-реальное";
    return block===0?"Я-реальное":"Я-идеальное";
  }

  function expandLearyLists(headers,rows){
    const nonStageCount=headers.filter(h=>!/^\s*(этап|волна|t0[–-]t1)\s*$/i.test(String(h??""))).length;if(nonStageCount>=128)return {headers,rows,expanded:false};
    const indexes=learyListColumns(headers);if(!indexes.length)return {headers,rows,expanded:false};
    const selected=v=>{const out=new Set(),s=String(v??"");for(const m of s.matchAll(/(?:^|[\n,])\s*(\d{1,3})\s*[.\)\t]/g)){const n=Number(m[1]);if(n>=1&&n<=128)out.add(n);}return out;};
    const label=index=>/идеаль/i.test(clean(headers[index]))?"Я-идеальное":/реаль/i.test(clean(headers[index]))?"Я-реальное":indexes.length===1?"Профиль":index===indexes[0]?"Я-реальное":"Я-идеальное";
    const extraHeaders=indexes.flatMap(index=>Array.from({length:128},(_,i)=>`${label(index)} / ${i+1}.`));
    return {headers:[...headers,...extraHeaders],rows:rows.map(r=>[...r,...indexes.flatMap(index=>{const answers=selected(r[index]);return Array.from({length:128},(_,i)=>answers.has(i+1)?1:0);})]),expanded:true};
  }

  function processTable({filename, headers:inputHeaders, rows:inputRows, methodId}) {
    let headers=inputHeaders,rows=inputRows;
    const method=methodId?byId[methodId]:detectMethod(filename,headers);
    if (!method) return {fatal:true,method:null,warnings:["Методика не распознана. Выберите ее вручную."],outputRows:[]};
    if(method.id==="leary_dmo_128"){const expanded=expandLearyLists(headers,rows);headers=expanded.headers;rows=expanded.rows;}
    const layout=chooseColumns(method,headers);
    if(layout.fatal)return {fatal:true,method,warnings:layout.warnings,outputRows:[]};
    const context=buildContext(method,rows,layout.indexes);
    const sexIndex=headers.findIndex(h=>/укажите\s+ваш\s+пол|^\s*пол\s*$/i.test(String(h??"")));
    const ageIndex=headers.findIndex(h=>/возраст/i.test(String(h??"")));
    const stageIndex=headers.findIndex(h=>/^\s*(?:этап|волна|t0[–-]t1)\s*$/i.test(String(h??"")));
    const idIndex=headers.findIndex(h=>/код.*(?:участник|респондент)|идентификатор.*(?:участник|респондент)|^\s*(?:id|ид)\s+(?:ответа|участника|респондента)\s*$|^\s*(?:e-?mail|email|фио|id|ид)\s*$|электронн.*почт/i.test(String(h??"")));
    const resultHeaders=[...headers]; const outputRows=[]; const allMetricNames=[]; const rowResults=[];
    for (const row of rows) {
      const blocks=[]; let validTotal=0; const rowWarnings=[]; let rowStatus="Обработано"; const combined={};
      for(let b=0;b<layout.blocks;b++){
        const idx=layout.indexes.slice(b*method.items,(b+1)*method.items);
        const raw=idx.map(i=>row[i]),x=raw.map(value=>normalizeValue(method,value,context));
        const valid=x.filter(v=>v!==null).length; validTotal+=valid;
        const unknown=raw.filter((value,index)=>!isBlank(value)&&x[index]===null).length;
        if(method.type==="passport"&&unknown){rowStatus="Неизвестные значения ответов";rowWarnings.push(`блок ${b+1}: неизвестных или недопустимых значений ${unknown}`);continue;}
        const passportPartial=method.type==="passport"&&method.missing?.mode==="scale";
        const required=passportPartial?0:method.type==="mean"?(method.minValid||method.items):method.items;
        if(valid<required){rowStatus="Неполные или некорректные ответы";rowWarnings.push(`блок ${b+1}: валидных ответов ${valid} из ${method.items}`);continue;}
        const scored=scoreComplete(method,x,{blockLabel:blockLabel(method,headers,layout.indexes,b),sexValue:sexIndex>=0?row[sexIndex]:null,ageValue:ageIndex>=0?row[ageIndex]:null});
        Object.assign(combined,scored.metrics); rowWarnings.push(...scored.warnings); if(scored.status!=="Обработано")rowStatus=scored.status;
      }
      for(const k of Object.keys(combined))if(!allMetricNames.includes(k))allMetricNames.push(k);
      rowResults.push({combined,validTotal,rowStatus,rowWarnings:[...new Set(rowWarnings)]});
    }
    const metaHeaders=["Методика","Статус обработки","Валидных ответов","Предупреждения"];
    resultHeaders.push(...metaHeaders,...allMetricNames);
    const excludedRows=[],allOutputRows=[];
    for(let r=0;r<rows.length;r++){
      const rr=rowResults[r],out=[...rows[r],method.name,rr.rowStatus,rr.validTotal,[...new Set([...layout.warnings,...rr.rowWarnings])].join("; "),...allMetricNames.map(k=>rr.combined[k]??null)];
      allOutputRows.push(out);
      if(rr.rowStatus==="Обработано")outputRows.push(out);
      else excludedRows.push({
        sourceRow:r+2,
        participantId:idIndex>=0?rows[r][idIndex]??"":"",
        stage:stageIndex>=0?rows[r][stageIndex]??"":"",
        reason:rr.rowStatus,

        validityScore:rr.combined["Достоверность – сырой балл"]??"",
        warnings:[...new Set([...layout.warnings,...rr.rowWarnings])].join("; ")
      });
    }
    const counts={
      total:rowResults.length,
      included:outputRows.length,
      excluded:excludedRows.length,
      incomplete:rowResults.filter(r=>r.rowStatus==="Неполные или некорректные ответы").length,
      unreliable:rowResults.filter(r=>r.rowStatus==="Недостоверный протокол").length,
      processed:outputRows.length,
      problems:excludedRows.length
    };
    return {fatal:false,method,headers:resultHeaders,outputRows,allOutputRows,excludedRows,warnings:layout.warnings,counts,passportColumns:layout.passportIndexes.length,responseColumns:layout.indexes.length,source:method.source};
  }
  return {methods,byId,detectMethod,chooseColumns,processTable,scoreComplete,extractQuestionNumber,normalizeForced,normalizeBinary};
